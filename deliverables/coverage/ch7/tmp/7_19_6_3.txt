[2] The printf function is equivalent to fprintf with the argument stdout interposed before the arguments to printf.

[3] The printf function returns the number of characters transmitted, or a negative value if an output or encoding error occurred.

#include <stdio.h>
#include <string.h>
#include <assert.h>
#include <stdarg.h>

/* ============================================================
 * 正向测试：以下代码应能编译并运行通过
 * ============================================================ */

/* [1] 原型检查：printf 的声明必须与标准一致。
 *     通过取函数指针类型来静态验证签名：
 *     int (*)(const char * restrict, ...)
 *     若原型不匹配，赋值将产生编译期诊断。 */
static int (*printf_proto_check)(const char * restrict, ...) = printf;

/* [2] 语义：printf(...) 等价于 fprintf(stdout, ...)。
 *     用一个辅助函数把输出重定向到临时文件，比较两者结果。 */
static int helper_printf_equiv(const char *fmt, ...)
{
    va_list ap;
    int n;
    va_start(ap, fmt);
    /* 直接调用 printf，验证其行为与 fprintf(stdout,...) 一致 */
    n = vprintf(fmt, ap);
    va_end(ap);
    return n;
}

int main(void)
{
    int n;

    /* ---------- [1] 原型与基本调用 ---------- */
    /* 函数指针类型匹配（编译期检查） */
    assert(printf_proto_check != NULL);

    /* 基本调用：格式串为普通字符串字面量 */
    n = printf("hello\n");
    assert(n == 6);                 /* "hello\n" 共 6 个字符 */

    /* ---------- [2] 语义：printf 等价于 fprintf(stdout, ...) ---------- */
    /* 用 vprintf 走同一路径，验证返回值语义一致 */
    n = helper_printf_equiv("abc%d\n", 42);
    assert(n == 6);                 /* "abc42\n" 共 6 个字符 */

    /* 空格式串：传输 0 个字符，返回 0 */
    n = printf("");
    assert(n == 0);

    /* 格式串中的转换说明：%d %s %c %% */
    n = printf("%d %s %c %%\n", -7, "xy", 'Z');
    /* 输出: "-7 xy Z %\n" => 10 个字符 */
    assert(n == 10);

    /* 宽度与精度 */
    n = printf("[%5d][%-5d][%.3s]\n", 42, 42, "abcdef");
    /* "[   42][42   ][abc]\n" => 1+5+1+1+5+1+1+3+1+1 = 20 */
    assert(n == 20);

    /* 长字符串：验证返回值等于实际字符数 */
    {
        const char *s = "0123456789";
        n = printf("%s\n", s);
        assert(n == 11);            /* 10 + '\n' */
    }

    /* ---------- [3] 返回值：非负整数，等于传输字符数 ---------- */
    /* 返回值类型为 int，且对成功输出应为非负 */
    n = printf("x");
    assert(n == 1);
    assert(n >= 0);

    /* 多次调用返回值累加应等于总字符数 */
    {
        int a = printf("aa");
        int b = printf("bbb");
        int c = printf("c");
        assert(a == 2 && b == 3 && c == 1);
        assert(a + b + c == 6);
    }

    /* 换行与制表符也计入字符数 */
    n = printf("\t\n");
    assert(n == 2);

    /* 格式串为运行时构造的字符串（非字面量） */
    {
        char fmt[16];
        strcpy(fmt, "%d\n");
        n = printf(fmt, 12345);
        assert(n == 6);             /* "12345\n" */
    }

    /* 使用 restrict 限定符的格式串参数：普通指针即可传入 */
    {
        const char * restrict rf = "ok\n";
        n = printf(rf);
        assert(n == 3);
    }

    printf("All positive tests passed.\n");
    return 0;
}
