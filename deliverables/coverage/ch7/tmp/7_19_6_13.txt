=== TEXT ===
The vsprintf function
Synopsis:
[1] #include <stdarg.h> #include <stdio.h> int vsprintf(char * restrict s, const char * restrict format, va_list arg);
Description:
[2] The vsprintf function is equivalent to sprintf, with the variable argument list replaced by arg, which shall have been initialized by the va_start macro (and possibly subsequent va_arg calls). The vsprintf function does not invoke the va_end macro. If copying takes place between objects that overlap, the behavior is undefined.
Returns:
[3] The vsprintf function returns the number of characters written in the array, not counting the terminating null character, or a negative value if an encoding error occurred.

#include <stdio.h>
#include <stdarg.h>
#include <string.h>
#include <assert.h>

/* 辅助函数：用 vsprintf 实现一个可变参数格式化函数 */
static int my_vsprintf(char *restrict buf, const char *restrict fmt, ...)
{
    va_list ap;
    int n;
    va_start(ap, fmt);          /* [2] arg 须由 va_start 初始化 */
    n = vsprintf(buf, fmt, ap); /* [1] 调用 vsprintf */
    va_end(ap);                 /* [2] vsprintf 本身不调用 va_end，由调用者负责 */
    return n;
}

/* 辅助函数：先消费一个 int 实参，再用剩余实参调用 vsprintf，
 * 验证「可能经过若干 va_arg 调用」后 arg 仍可用于 vsprintf */
static int consume_then_vsprintf(char *restrict buf, const char *restrict fmt, ...)
{
    va_list ap;
    int first;
    int n;
    va_start(ap, fmt);
    first = va_arg(ap, int);    /* [2] 先做一次 va_arg */
    n = vsprintf(buf, fmt, ap); /* [2] 之后 arg 仍可传给 vsprintf */
    va_end(ap);
    (void)first;
    return n;
}

int main(void)
{
    /* ========== 正向测试：以下代码应能编译并运行通过 ========== */
    {
        char buf[128];
        int n;

        /* [1][2][3] 基本用法：格式化字符串，返回写入字符数（不含 '\0'） */
        n = my_vsprintf(buf, "hello %s, %d!", "world", 42);
        assert(n == (int)strlen("hello world, 42!"));
        assert(strcmp(buf, "hello world, 42!") == 0);

        /* [3] 返回值等于写入数组的字符数，不含结尾空字符 */
        n = my_vsprintf(buf, "%d", 12345);
        assert(n == 5);
        assert(strcmp(buf, "12345") == 0);
        assert(buf[5] == '\0');   /* 结尾空字符存在但不计入返回值 */

        /* [3] 空格式串：写入 0 个字符，返回 0 */
        n = my_vsprintf(buf, "");
        assert(n == 0);
        assert(buf[0] == '\0');

        /* [2] 等价于 sprintf：与 sprintf 结果逐字节比较 */
        {
            char a[128], b[128];
            int na, nb;
            na = my_vsprintf(a, "x=%d y=%s z=%c", -7, "abc", 'Q');
            nb = sprintf(b, "x=%d y=%s z=%c", -7, "abc", 'Q');
            assert(na == nb);
            assert(strcmp(a, b) == 0);
        }

        /* [2] arg 经过若干 va_arg 调用后仍可传给 vsprintf */
        n = consume_then_vsprintf(buf, "rest=%d,%d", 111, 222, 333);
        assert(n == (int)strlen("rest=222,333"));
        assert(strcmp(buf, "rest=222,333") == 0);

        /* [2] 多个可变实参、多种转换说明 */
        n = my_vsprintf(buf, "%s|%d|%u|%x|%c|%f",
                        "s", -1, 4000000000u, 255, 'A', 1.5);
        assert(strcmp(buf, "s|-1|4000000000|ff|A|1.500000") == 0);
        assert(n == (int)strlen(buf));

        /* [2] 宽度/精度等格式控制与 sprintf 一致 */
        n = my_vsprintf(buf, "[%5d][%-5d][%.3s]", 42, 42, "abcdef");

                    int y = va_arg(ap, int);   /* vsprintf 未调用 va_end，arg 仍有效 */
                    assert(y == 999);
                }
            }
            va_end(ap);
        }

    }
