--- TEXT ---
The strcpy function
Synopsis:
[1] #include <string.h> char *strcpy(char * restrict s1, const char * restrict s2);
Description:
[2] The strcpy function copies the string pointed to by s2 (including the terminating null character) into the array pointed to by s1. If copying takes place between objects that overlap, the behavior is undefined.
Returns:
[3] The strcpy function returns the value of s1.

 *   [3] 返回值：返回 s1 的值。

#include <stdio.h>
#include <string.h>
#include <assert.h>

/* ========== 正向测试：以下代码应能编译并运行通过 ========== */

/* [1] 原型存在性检查：把 strcpy 的地址赋给匹配的函数指针类型。
 *     若原型与标准不符（参数类型/返回类型/const 限定），此处会告警或报错。 */
static char *(*fp_strcpy)(char * restrict, const char * restrict) = strcpy;

int main(void)
{
    /* [2] 基本复制：s2 的字符串（含结尾 '\0'）被复制到 s1 指向的数组 */
    {
        char dst[16];
        const char *src = "hello";
        char *ret;

        /* 先把 dst 填满非零字节，验证结尾 '\0' 确实被复制过来 */
        memset(dst, 'X', sizeof dst);

        ret = strcpy(dst, src);

        assert(strcmp(dst, "hello") == 0);   /* 内容正确 */
        assert(dst[5] == '\0');              /* 结尾空字符被复制 */
        assert(dst[6] == 'X');               /* 只复制到 '\0' 为止，未越界写 */

        /* [3] 返回值等于 s1 */
        assert(ret == dst);
        assert(ret == &dst[0]);
    }

    /* [2] 复制空字符串：只写入一个 '\0' */
    {
        char dst[8];
        char *ret;

        memset(dst, 'Y', sizeof dst);
        ret = strcpy(dst, "");

        assert(dst[0] == '\0');
        assert(dst[1] == 'Y');               /* 仅写了一个字节 */
        assert(ret == dst);                  /* [3] */
    }

    /* [2] 源为 const 限定对象：形参是 const char *，应能接受 const 实参 */
    {
        const char src_const[] = "const-src";
        char dst[16];
        char *ret = strcpy(dst, src_const);

        assert(strcmp(dst, "const-src") == 0);
        assert(ret == dst);                  /* [3] */
    }

    /* [2] 目标为数组元素起始处（非数组名），验证 s1 可为任意 char* 左值 */
    {
        char buf[32];
        char *p = buf + 4;
        char *ret = strcpy(p, "abc");

        assert(strcmp(buf + 4, "abc") == 0);
        assert(ret == p);                    /* [3] 返回传入的指针本身 */
    }

    /* [3] 返回值可用于链式调用：strcpy 的返回值作为另一个 strcpy 的 s1 */
    {
        char a[16], b[16];
        char *ret = strcpy(a, strcpy(b, "chain"));

        assert(strcmp(b, "chain") == 0);
        assert(strcmp(a, "chain") == 0);
        assert(ret == a);
    }

    /* [1] 通过函数指针调用，验证原型签名可用 */
    {
        char dst[16];
        char *ret = fp_strcpy(dst, "via-fp");

        assert(strcmp(dst, "via-fp") == 0);
        assert(ret == dst);                  /* [3] */
    }

    /* [2] 复制后源字符串保持不变（源是 const，语义上不应被修改） */
    {
        char src[] = "immutable";
        char dst[16];
        strcpy(dst, src);

        assert(strcmp(src, "immutable") == 0);
        assert(strcmp(dst, "immutable") == 0);
    }

    printf("All positive tests for C99 7.21.2.3 strcpy passed.\n");
    return 0;
}
