=== ARTICLE 7.21.5.7 ===
--- TEXT ---
The strstr function
Synopsis:
[1] #include <string.h> char *strstr(const char *s1, const char *s2);
Description:
[2] The strstr function locates the first occurrence in the string pointed to by s1 of the sequence of characters (excluding the terminating null character) in the string pointed to by s2.
Returns:
[3] The strstr function returns a pointer to the located string, or a null pointer if the string is not found. If s2 points to a string with zero length, the function returns s1.
--- CODE ---

 *   [3] 返回值：找到则返回指向该处的指针；未找到返回空指针；
 *              若 s2 指向零长度字符串，则返回 s1。

#include <stdio.h>
#include <string.h>
#include <assert.h>

/* ========== 正向测试：以下代码应能编译并运行通过 ========== */

/* [1] 原型检查：返回类型为 char *，参数为两个 const char *。
 *     通过函数指针赋值来静态验证签名（不匹配则编译报错）。 */
static char *(*strstr_sig)(const char *, const char *) = strstr;

int main(void)
{
    /* [1] 原型可用性：调用形式与标准一致 */
    const char *hay = "hello world, hello C99";
    const char *needle = "hello";

    /* [2] 定位首次出现：应指向第一个 "hello" */
    char *p = strstr(hay, needle);
    assert(p != NULL);
    assert(p == hay);                 /* 首次出现就在开头 */
    assert(strncmp(p, "hello", 5) == 0);

    /* [2] 定位中间位置的首次出现 */
    const char *s1 = "abcabcabc";
    char *q = strstr(s1, "bca");
    assert(q != NULL);
    assert(q == s1 + 1);              /* 第一个 "bca" 从索引 1 开始 */

    /* [2] 只匹配完整序列，不匹配部分 */
    char *r = strstr("abcdef", "xyz");
    assert(r == NULL);                /* 未找到 */

    /* [3] 未找到时返回空指针 */
    assert(strstr("foobar", "baz") == NULL);

    /* [3] s2 为零长度字符串时返回 s1 */
    const char *empty = "";
    char *e = strstr(hay, empty);
    assert(e == hay);                 /* 必须返回 s1 本身 */

    /* [3] 零长度 s2 对空 s1 也返回 s1 */
    const char *empty_s1 = "";
    char *e2 = strstr(empty_s1, empty);
    assert(e2 == empty_s1);

    /* [2][3] s2 等于整个 s1 时返回 s1 */
    char *full = strstr("exact", "exact");
    assert(full != NULL);
    assert(strcmp(full, "exact") == 0);

    /* [2] 终止空字符不参与匹配：s2 含 '\0' 之外的内容 */
    char *tail = strstr("abc", "c");
    assert(tail != NULL && *tail == 'c' && tail[1] == '\0');

    /* [1] 通过函数指针调用，验证签名一致 */
    char *via_ptr = strstr_sig("pointer call", "call");
    assert(via_ptr != NULL && strcmp(via_ptr, "call") == 0);

    /* [2] 返回的指针可用于修改（s1 为可写数组时） */
    char buf[] = "one two one";
    char *m = strstr(buf, "one");
    assert(m == buf);
    m[0] = 'O';                       /* 修改定位到的字符 */
    assert(strcmp(buf, "One two one") == 0);

    printf("All positive tests for C99 7.21.5.7 strstr passed.\n");
    return 0;
}
