=== ARTICLE 7.21.4 ===
--- TEXT ---
Comparison functions
[1] The sign of a nonzero value returned by the comparison functions memcmp, strcmp, and strncmp is determined by the sign of the difference between the values of the first pair of characters (both interpreted as unsigned char) that differ in the objects being compared.
--- CODE ---
/*
 * 测试 C99 7.21.4 [1]：比较函数 memcmp / strcmp / strncmp 的返回值符号
 * 由「第一对不同的字符（均按 unsigned char 解释）之差」的符号决定。
 *
 * 预期行为：
 *   正向测试：以下代码应能编译并运行通过（assert 全部成立）。
 *   负向测试：违反约束的片段应导致编译报错（统一放在 #if 0 中）。
 *
 * 说明：本条款只规定返回值的「符号」，不规定具体数值，因此测试只检查
 *       < 0、== 0、> 0 三种符号关系，不检查具体返回值大小。
 */

#include <stdio.h>
#include <string.h>
#include <assert.h>

/* ========== 正向测试：以下代码应能编译并运行通过 ========== */

/* 辅助：把比较结果归一化为符号 -1 / 0 / +1 */
static int sgn(int v)
{
    return (v > 0) - (v < 0);
}

int main(void)
{
    /* ---------------------------------------------------------------
     * [1] memcmp：符号由第一对不同字符（按 unsigned char 解释）之差决定
     * --------------------------------------------------------------- */

    /* [1] 完全相同 -> 返回 0 */
    {
        unsigned char a[] = { 0x00, 0x01, 0x02, 0x03 };
        unsigned char b[] = { 0x00, 0x01, 0x02, 0x03 };
        assert(memcmp(a, b, sizeof a) == 0);
    }

    /* [1] 第一对不同字符：a 的 0x01 < b 的 0x02 -> 负 */
    {
        unsigned char a[] = { 0x00, 0x01, 0x02 };
        unsigned char b[] = { 0x00, 0x02, 0x02 };
        assert(sgn(memcmp(a, b, sizeof a)) < 0);
        assert(sgn(memcmp(b, a, sizeof a)) > 0);
    }

    /* [1] 关键点：按 unsigned char 解释，而非 signed char。
     *     0xFF 作为 unsigned char 是 255，大于 0x01 的 1。
     *     若错误地按 signed char 比较，0xFF 会被当作 -1，结果符号相反。 */
    {
        unsigned char a[] = { 0xFF };
        unsigned char b[] = { 0x01 };
        /* 255 - 1 = 254 > 0，故 a > b */
        assert(sgn(memcmp(a, b, 1)) > 0);
        assert(sgn(memcmp(b, a, 1)) < 0);
    }

    /* [1] 0x80 (128) 与 0x7F (127)：unsigned 解释下 128 > 127 */
    {
        unsigned char a[] = { 0x80 };
        unsigned char b[] = { 0x7F };
        assert(sgn(memcmp(a, b, 1)) > 0);
        assert(sgn(memcmp(b, a, 1)) < 0);
    }

    /* [1] 只有「第一对」不同字符决定符号：后续字符的差异不影响符号 */
    {
        unsigned char a[] = { 0x01, 0xFF, 0xFF };
        unsigned char b[] = { 0x02, 0x00, 0x00 };
        /* 第一对不同：0x01 < 0x02 -> 负，尽管后面 a 更大 */
        assert(sgn(memcmp(a, b, sizeof a)) < 0);
    }

    /* [1] n == 0 时无字符比较 -> 返回 0 */
    {
        unsigned char a[] = { 0x01 };
        unsigned char b[] = { 0x02 };
        assert(memcmp(a, b, 0) == 0);
    }

    /* ---------------------------------------------------------------
     * [1] strcmp：符号由第一对不同字符（按 unsigned char 解释）之差决定
     * --------------------------------------------------------------- */

    /* [1] 相同字符串 -> 0 */
    assert(strcmp("hello", "hello") == 0);
    assert(strcmp("", "") == 0);

    /* [1] 第一对不同字符决定符号 */
    assert(sgn(strcmp("abc", "abd")) < 0);
    assert(sgn(strcmp("abd", "abc")) > 0);

    /* [1] 前缀关系：短串在 '\0' 处与长串字符比较，'\0' 更小 -> 负 */
    assert(sgn(strcmp("abc", "abcd")) < 0);
    assert(sgn(strcmp("abcd", "abc")) > 0);

    /* [1] 按 unsigned char 解释：'\xFF' (255) > 'a' (97) */
    {
        char a[] = { (char)0xFF, '\0' };
        char b[] = { 'a', '\0' };
        assert(sgn(strcmp(a, b)) > 0);
        assert(sgn(strcmp(b, a)) < 0);
    }

    /* [1] 按 unsigned char 解释：'\x80' (128) > '\x7F' (127) */
    {
        char a[] = { (char)0x80, '\0' };
        char b[] = { (char)0x7F, '\0' };
        assert(sgn(strcmp(a, b)) > 0);
        assert(sgn(strcmp(b, a)) < 0);
    }

    /* [1] 只有第一对不同字符决定符号 */
    assert(sgn(strcmp("aZZZ", "bAAA")) < 0);

    /* ---------------------------------------------------------------
     * [1] strncmp：符号由第一对不同字符（按 unsigned char 解释）之差决定
     * --------------------------------------------------------------- */

    /* [1] 前 n 个字符相同 -> 0 */
    assert(strncmp("hello", "help!", 3) == 0);
    assert(strncmp("abc", "abd", 2) == 0);

    /* [1] 第一对不同字符决定符号 */
    assert(sgn(strncmp("abc", "abd", 3)) < 0);
    assert(sgn(strncmp("abd", "abc", 3)) > 0);

    /* [1] n 限制比较长度：差异在 n 之外则返回 0 */
    assert(strncmp("abcX", "abcY", 3) == 0);

    /* [1] 前缀关系：在 n 范围内遇到 '\0'，'\0' 更小 -> 负 */
    assert(sgn(strncmp("abc", "abcd", 4)) < 0);
    assert(sgn(strncmp("abcd", "abc", 4)) > 0);

    /* [1] 按 unsigned char 解释 */
    {
        char a[] = { (char)0xFF, '\0' };
        char b[] = { 'a', '\0' };
        assert(sgn(strncmp(a, b, 1)) > 0);
        assert(sgn(strncmp(b, a, 1)) < 0);
    }

    /* [1] n == 0 -> 0 */
    assert(strncmp("abc", "xyz", 0) == 0);

    /* [1] 只有第一对不同字符决定符号 */
    assert(sgn(strncmp("aZZZ", "bAAA", 4)) < 0);

    printf("All positive tests passed.\n");
    return 0;
}
