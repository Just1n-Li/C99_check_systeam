=== TEXT ===
The wmemset function
Synopsis:
[1] #include <wchar.h> wchar_t *wmemset(wchar_t *s, wchar_t c, size_t n);
Description:
[2] The wmemset function copies the value of c into each of the first n wide characters of the object pointed to by s.
Returns:
[3] The wmemset function returns the value of s.

=== CODE ===

 *   [1] 原型：wchar_t *wmemset(wchar_t *s, wchar_t c, size_t n);
 *   [2] 语义：把 c 的值复制到 s 所指对象的前 n 个宽字符中。
 *   [3] 返回值：返回 s 的值。

#include <stdio.h>
#include <wchar.h>
#include <stddef.h>
#include <assert.h>

/* ========== 正向测试：以下代码应能编译并运行通过 ========== */

/* [1] 原型检查：函数指针类型必须与标准原型完全一致 */
static wchar_t *(*fp_wmemset)(wchar_t *, wchar_t, size_t) = wmemset;

int main(void)
{
    /* [1] 原型可用性：能取地址、能通过函数指针调用 */
    assert(fp_wmemset == wmemset);

    /* [2] 基本语义：把 c 复制到前 n 个宽字符 */
    {
        wchar_t buf[8] = { L'A', L'B', L'C', L'D', L'E', L'F', L'G', L'H' };
        wchar_t *r;

        r = wmemset(buf, L'Z', 3);
        assert(buf[0] == L'Z');
        assert(buf[1] == L'Z');
        assert(buf[2] == L'Z');
        /* 第 n 个之后必须保持不变 */
        assert(buf[3] == L'D');
        assert(buf[4] == L'E');
        assert(buf[5] == L'F');
        assert(buf[6] == L'G');
        assert(buf[7] == L'H');

        /* [3] 返回值必须等于 s */
        assert(r == buf);
    }

    /* [2] n == 0：不写入任何字符，缓冲区保持不变 */
    {
        wchar_t buf[4] = { L'1', L'2', L'3', L'4' };
        wchar_t *r = wmemset(buf, L'X', 0);
        assert(buf[0] == L'1');
        assert(buf[1] == L'2');
        assert(buf[2] == L'3');
        assert(buf[3] == L'4');
        /* [3] 即使 n == 0 也返回 s */
        assert(r == buf);
    }

    /* [2] 覆盖整个数组 */
    {
        wchar_t buf[5] = { L'a', L'b', L'c', L'd', L'e' };
        wchar_t *r = wmemset(buf, L'\0', 5);
        int i;
        for (i = 0; i < 5; i++)
            assert(buf[i] == L'\0');
        assert(r == buf);
    }

    /* [2] 复制宽字符 L'\0' 之外的值，例如非 ASCII 宽字符 */
    {
        wchar_t buf[3] = { 0, 0, 0 };
        wchar_t *r = wmemset(buf, (wchar_t)0x4E2D /* 中 */, 3);
        assert(buf[0] == (wchar_t)0x4E2D);
        assert(buf[1] == (wchar_t)0x4E2D);
        assert(buf[2] == (wchar_t)0x4E2D);
        assert(r == buf);
    }

    /* [2] 从数组中间开始填充（s 指向对象内部） */
    {
        wchar_t buf[6] = { L'0', L'1', L'2', L'3', L'4', L'5' };
        wchar_t *r = wmemset(buf + 2, L'#', 3);
        assert(buf[0] == L'0');
        assert(buf[1] == L'1');
        assert(buf[2] == L'#');
        assert(buf[3] == L'#');
        assert(buf[4] == L'#');
        assert(buf[5] == L'5');
        /* [3] 返回的是传入的 s（即 buf + 2），不是 buf */
        assert(r == buf + 2);
    }

    /* [2] 用 wmemset 构造宽字符串，再用 wcslen 验证 */
    {
        wchar_t buf[10];
        wmemset(buf, L'q', 9);
        buf[9] = L'\0';
        assert(wcslen(buf) == 9);
        assert(buf[0] == L'q' && buf[8] == L'q');
    }

    /* [3] 返回值可链式使用 */
    {
        wchar_t buf[4] = { 0, 0, 0, 0 };
        wchar_t *r = wmemset(wmemset(buf, L'A', 2), L'B', 2);
        assert(r == buf);
        assert(buf[0] == L'B');
        assert(buf[1] == L'B');
        assert(buf[2] == L'A');
        assert(buf[3] == L'A');
    }

    printf("wmemset: all positive tests passed.\n");
    return 0;
}
