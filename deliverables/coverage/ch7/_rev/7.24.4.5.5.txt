=== TEXT ===
The wcsspn function
Synopsis:
[1] #include <wchar.h> size_t wcsspn(const wchar_t *s1, const wchar_t *s2);
Description:
[2] The wcsspn function computes the length of the maximum initial segment of the wide string pointed to by s1 which consists entirely of wide characters from the wide string pointed to by s2.
Returns:
[3] The wcsspn function returns the length of the segment.

=== CODE ===

 *   负向测试：违反约束的代码（如参数类型错误、缺少头文件声明等）应导致
 *             编译报错。这些片段放在 #if 0 中，不影响本文件正常编译。

 * 覆盖段落：
 *   [1] 函数原型 / 头文件
 *   [2] 语义：计算最长初始段长度
 *   [3] 返回值：返回该段长度

#include <wchar.h>
#include <stdio.h>
#include <assert.h>
#include <stddef.h>

/* ========== 正向测试：以下代码应能编译并运行通过 ========== */

/* [1] 验证函数原型可用：size_t wcsspn(const wchar_t *, const wchar_t *); */
static size_t (*fp)(const wchar_t *, const wchar_t *) = wcsspn;

int main(void)
{
    /* [1] 头文件 <wchar.h> 已包含，函数可被调用 */
    assert(fp == wcsspn);

    /* [2][3] 基本语义：s1 的初始段全部由 s2 中的字符组成 */
    {
        const wchar_t *s1 = L"abcdeXYZ";
        const wchar_t *s2 = L"abcde";
        size_t n = wcsspn(s1, s2);
        /* "abcde" 全部在 s2 中，'X' 不在，故长度为 5 */
        assert(n == 5);
        printf("[2][3] wcsspn(L\"abcdeXYZ\", L\"abcde\") = %zu (期望 5)\n", n);
    }

    /* [2][3] s1 的第一个字符就不在 s2 中，返回 0 */
    {
        const wchar_t *s1 = L"XYZabc";
        const wchar_t *s2 = L"abc";
        size_t n = wcsspn(s1, s2);
        assert(n == 0);
        printf("[2][3] wcsspn(L\"XYZabc\", L\"abc\") = %zu (期望 0)\n", n);
    }

    /* [2][3] s1 全部由 s2 中字符组成，返回 s1 的完整长度 */
    {
        const wchar_t *s1 = L"aabbcc";
        const wchar_t *s2 = L"abc";
        size_t n = wcsspn(s1, s2);
        assert(n == 6);
        printf("[2][3] wcsspn(L\"aabbcc\", L\"abc\") = %zu (期望 6)\n", n);
    }

    /* [2][3] s1 为空宽字符串，最长初始段长度为 0 */
    {
        const wchar_t *s1 = L"";
        const wchar_t *s2 = L"abc";
        size_t n = wcsspn(s1, s2);
        assert(n == 0);
        printf("[2][3] wcsspn(L\"\", L\"abc\") = %zu (期望 0)\n", n);
    }

    /* [2][3] s2 为空宽字符串，任何非空 s1 的初始段长度均为 0 */
    {
        const wchar_t *s1 = L"abc";
        const wchar_t *s2 = L"";
        size_t n = wcsspn(s1, s2);
        assert(n == 0);
        printf("[2][3] wcsspn(L\"abc\", L\"\") = %zu (期望 0)\n", n);
    }

    /* [2][3] s1 与 s2 均为空，返回 0 */
    {
        const wchar_t *s1 = L"";
        const wchar_t *s2 = L"";
        size_t n = wcsspn(s1, s2);
        assert(n == 0);
        printf("[2][3] wcsspn(L\"\", L\"\") = %zu (期望 0)\n", n);
    }

    /* [2][3] s2 含重复字符，不影响结果 */
    {
        const wchar_t *s1 = L"aabbccdd";
        const wchar_t *s2 = L"aabbccdd";
        size_t n = wcsspn(s1, s2);
        assert(n == 8);
        printf("[2][3] wcsspn(L\"aabbccdd\", L\"aabbccdd\") = %zu (期望 8)\n", n);
    }

    /* [2][3] 非 ASCII 宽字符（如中文）也应正确处理 */
    {
        const wchar_t *s1 = L"\u4f60\u597d\u4e16\u754cXYZ";
        const wchar_t *s2 = L"\u4f60\u597d\u4e16\u754c";
        size_t n = wcsspn(s1, s2);
        assert(n == 4);
        printf("[2][3] wcsspn(中文前缀) = %zu (期望 4)\n", n);
    }

    /* [2][3] s1 中段出现不在 s2 中的字符，只统计初始段 */
    {
        const wchar_t *s1 = L"aaabbbccc";
        const wchar_t *s2 = L"ab";
        size_t n = wcsspn(s1, s2);
        /* "aaabbb" 全部由 'a','b' 组成，'c' 不在 s2 中，长度 6 */
        assert(n == 6);
        printf("[2][3] wcsspn(L\"aaabbbccc\", L\"ab\") = %zu (期望 6)\n", n);
    }

    /* [2][3] 返回值类型为 size_t，可与其他 size_t 比较 */
    {
        const wchar_t *s1 = L"hello";
        const wchar_t *s2 = L"hel";
        size_t n = wcsspn(s1, s2);
        assert(n == 3);
        assert(n == (size_t)3);
        printf("[2][3] wcsspn(L\"hello\", L\"hel\") = %zu (期望 3)\n", n);
    }

    printf("所有正向测试通过。\n");
    return 0;
}
